示例一：一条完整训练链路

对应第 1–7 章：用二维合成数据训练小型分类网络，包含 Dataset、DataLoader、训练模式、清梯度、前向、Loss、反向和参数更新。

下载 train.py

"""CPU-friendly classification demo: python examples/train.py"""
import torch
from torch import nn
from torch.utils.data import DataLoader, TensorDataset


def main():
    torch.manual_seed(42)
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    x = torch.randn(256, 2)
    y = (x[:, 0] * x[:, 1] > 0).long()
    loader = DataLoader(TensorDataset(x, y), batch_size=32, shuffle=True)
    model = nn.Sequential(nn.Linear(2, 16), nn.ReLU(), nn.Linear(16, 2)).to(device)
    loss_fn = nn.CrossEntropyLoss()
    optimizer = torch.optim.Adam(model.parameters(), lr=0.02)
    for epoch in range(60):
        model.train()
        total_loss = 0.0
        for features, labels in loader:
            features, labels = features.to(device), labels.to(device)
            optimizer.zero_grad(set_to_none=True)
            loss = loss_fn(model(features), labels)
            loss.backward()
            optimizer.step()
            total_loss += loss.item() * len(features)
        if (epoch + 1) % 10 == 0:
            print(f'Epoch {epoch + 1}: loss={total_loss / len(x):.4f}')
    model.eval()
    with torch.no_grad():
        accuracy = (model(x.to(device)).argmax(dim=-1) == y.to(device)).float().mean()
    print(f'Training accuracy (not a test-set metric): {accuracy.item():.1%}')


if __name__ == '__main__':
    main()

示例二：可以运行的 Transformer Block

对应第 9–10 章：支持双向和因果模式，包含反向传播与“未来输入不影响过去输出”的因果性检查。运行后应输出 torch.Size([2, 10, 128]) 和检查通过提示。它仍不是完整 GPT。

"""Minimal Pre-Norm block; python examples/transformer.py."""
import torch
from torch import nn


class TransformerBlock(nn.Module):
    def __init__(self, d_model=128, num_heads=4):
        super().__init__()
        self.ln1 = nn.LayerNorm(d_model)
        self.attn = nn.MultiheadAttention(d_model, num_heads, batch_first=True)
        self.ln2 = nn.LayerNorm(d_model)
        self.ffn = nn.Sequential(
            nn.Linear(d_model, 4 * d_model),
            nn.GELU(),
            nn.Linear(4 * d_model, d_model),
        )

    def forward(self, x, causal=False, padding_mask=None):
        h = self.ln1(x)
        mask = None
        if causal:
            n = x.size(1)
            mask = torch.triu(torch.ones(n, n, device=x.device, dtype=torch.bool), diagonal=1)
        out, _ = self.attn(h, h, h, attn_mask=mask,
                           key_padding_mask=padding_mask, need_weights=False)
        x = x + out
        return x + self.ffn(self.ln2(x))


if __name__ == '__main__':
    torch.manual_seed(42)
    block = TransformerBlock()
    x = torch.randn(2, 10, 128)
    y = block(x, causal=True)
    print(y.shape)
    y.square().mean().backward()
    assert all(p.grad is not None and torch.isfinite(p.grad).all() for p in block.parameters())
    block.eval()
    with torch.no_grad():
        changed = x.clone()
        changed[:, 6:] += 10
        assert torch.allclose(block(x, causal=True)[:, :6], block(changed, causal=True)[:, :6], atol=1e-6)
    print('Backward pass and causal-mask check passed.')

"""Chapter 11: small character-level GPT. No dataset/model downloads required."""
import argparse
import torch
from torch import nn
import torch.nn.functional as F


class TransformerBlock(nn.Module):
    def __init__(self, d_model, num_heads):
        super().__init__()
        self.ln1 = nn.LayerNorm(d_model)
        self.attn = nn.MultiheadAttention(d_model, num_heads, batch_first=True)
        self.ln2 = nn.LayerNorm(d_model)
        self.ffn = nn.Sequential(
            nn.Linear(d_model, 4 * d_model), nn.GELU(),
            nn.Linear(4 * d_model, d_model),
        )

    def forward(self, x):
        T = x.size(1)
        causal_mask = torch.triu(
            torch.ones(T, T, device=x.device, dtype=torch.bool), diagonal=1
        )
        h = self.ln1(x)
        out, _ = self.attn(h, h, h, attn_mask=causal_mask, need_weights=False)
        x = x + out
        return x + self.ffn(self.ln2(x))


class MiniGPT(nn.Module):
    def __init__(self, vocab_size, d_model=64, num_heads=4,
                 num_layers=2, block_size=32):
        super().__init__()
        self.block_size = block_size
        self.token_embedding = nn.Embedding(vocab_size, d_model)
        self.position_embedding = nn.Embedding(block_size, d_model)
        self.blocks = nn.Sequential(*[
            TransformerBlock(d_model, num_heads) for _ in range(num_layers)
        ])
        self.ln_f = nn.LayerNorm(d_model)
        self.lm_head = nn.Linear(d_model, vocab_size)

    def forward(self, idx, targets=None):
        B, T = idx.shape
        if not 0 < T <= self.block_size:
            raise ValueError('Sequence length must be in [1, block_size]')
        positions = torch.arange(T, device=idx.device)
        x = self.token_embedding(idx) + self.position_embedding(positions)
        logits = self.lm_head(self.ln_f(self.blocks(x)))
        loss = None
        if targets is not None:
            # targets are ALREADY shifted by get_batch; do not shift twice.
            loss = F.cross_entropy(logits.reshape(-1, logits.size(-1)), targets.reshape(-1))
        return logits, loss


@torch.no_grad()
def generate(model, idx, max_new_tokens, temperature=1.0, top_k=None, top_p=1.0):
    if temperature <= 0 or not 0 < top_p <= 1:
        raise ValueError('Require temperature > 0 and 0 < top_p <= 1')
    if top_k is not None and top_k < 1:
        raise ValueError('top_k must be positive')
    was_training = model.training
    model.eval()
    try:
        for _ in range(max_new_tokens):
            idx_cond = idx[:, -model.block_size:]
            logits, _ = model(idx_cond)
            logits = logits[:, -1, :] / temperature
            if top_k is not None:
                threshold = logits.topk(min(top_k, logits.size(-1))).values[:, [-1]]
                logits = logits.masked_fill(logits < threshold, float('-inf'))
            if top_p < 1:
                sorted_logits, order = logits.sort(descending=True, dim=-1)
                cumulative = sorted_logits.softmax(dim=-1).cumsum(dim=-1)
                remove = cumulative > top_p
                # Keep the first token crossing the threshold, including top-1.
                remove[:, 1:] = remove[:, :-1].clone()
                remove[:, 0] = False
                sorted_logits = sorted_logits.masked_fill(remove, float('-inf'))
                logits = torch.empty_like(logits).scatter(1, order, sorted_logits)
            next_token = torch.multinomial(logits.softmax(dim=-1), num_samples=1)
            idx = torch.cat([idx, next_token], dim=1)
    finally:
        model.train(was_training)
    return idx


def main(steps=200):
    torch.manual_seed(42)
    torch.set_num_threads(2)
    text = ('林默走进实验室。教授正在研究PyTorch。神经网络正在学习语言。'
            '林默问教授，Transformer为什么能够生成文本？'
            '教授回答，因为模型能够预测下一个Token。')
    chars = sorted(set(text))
    stoi = {ch: i for i, ch in enumerate(chars)}
    data = torch.tensor([stoi[ch] for ch in text], dtype=torch.long)
    block_size, batch_size = 32, 8
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    model = MiniGPT(len(chars), block_size=block_size).to(device)
    optimizer = torch.optim.AdamW(model.parameters(), lr=1e-3)
    model.train()
    for step in range(steps):
        starts = torch.randint(len(data) - block_size, (batch_size,))
        x = torch.stack([data[i:i + block_size] for i in starts]).to(device)
        y = torch.stack([data[i + 1:i + block_size + 1] for i in starts]).to(device)
        optimizer.zero_grad(set_to_none=True)
        _, loss = model(x, y)
        if not torch.isfinite(loss):
            raise RuntimeError('Non-finite training loss')
        loss.backward()
        optimizer.step()
        if step == 0 or (step + 1) % 50 == 0:
            print(f'Step {step + 1}: loss={loss.item():.4f}')
    context = torch.tensor([[stoi[c] for c in '林默']], device=device)
    output = generate(model, context, 60, temperature=0.8, top_k=10)
    print(''.join(chars[i] for i in output[0].tolist()))
    print('Toy corpus demonstration only; this is not a question-answering model.')


if __name__ == '__main__':
    parser = argparse.ArgumentParser()
    parser.add_argument('--steps', type=int, default=200)
    main(parser.parse_args().steps)

示例四：LoRA 的冻结与梯度

对应第十二章。验证初始输出不变、基础权重冻结，以及第一步 B 获得梯度、之后 A 参与学习。

"""Chapter 12: verify LoRA initialization and frozen base weights on CPU."""
import math
import torch
from torch import nn
import torch.nn.functional as F


class LoRALinear(nn.Module):
    def __init__(self, original_linear, rank=16, alpha=32):
        super().__init__()
        if rank < 1:
            raise ValueError('rank must be positive')
        self.original = original_linear
        for param in self.original.parameters():
            param.requires_grad_(False)
        self.scaling = alpha / rank
        # Follow the original layer's dtype and device.
        weight = original_linear.weight
        self.lora_A = nn.Parameter(weight.new_empty(rank, original_linear.in_features))
        self.lora_B = nn.Parameter(weight.new_zeros(original_linear.out_features, rank))
        nn.init.kaiming_uniform_(self.lora_A, a=math.sqrt(5))

    def forward(self, x):
        delta = F.linear(F.linear(x, self.lora_A), self.lora_B)
        return self.original(x) + self.scaling * delta


def main():
    torch.manual_seed(42)
    base = nn.Linear(128, 128)
    layer = LoRALinear(base, rank=8, alpha=16)
    x, target = torch.randn(4, 128), torch.randn(4, 128)
    original = {name: p.detach().clone() for name, p in base.named_parameters()}
    assert torch.equal(layer(x), base(x)), 'Zero B must initially preserve base output'
    optimizer = torch.optim.AdamW([p for p in layer.parameters() if p.requires_grad], lr=1e-3)
    for step in range(2):
        optimizer.zero_grad(set_to_none=True)
        loss = F.mse_loss(layer(x), target)
        loss.backward()
        assert all(p.grad is None for p in base.parameters())
        assert layer.lora_B.grad.abs().sum() > 0
        if step == 0:
            assert torch.count_nonzero(layer.lora_A.grad) == 0
        else:
            assert layer.lora_A.grad.abs().sum() > 0
        optimizer.step()
    assert all(torch.equal(p, original[name]) for name, p in base.named_parameters())
    print('PASS: initial identity, frozen base, B-first then A gradient flow.')


if __name__ == '__main__':
    main()

示例五：SFT 与 DPO 的目标

对应第十三章。验证首个回答 Token 对齐、Prompt/PAD 目标屏蔽、回答 logp 求和与 DPO 梯度方向。这是核心损失实验，不是完整 SFT/DPO 训练系统。

"""Chapter 13: shifted response-only SFT and classic DPO loss checks."""
import torch
import torch.nn.functional as F


def sft_loss(logits, labels):
    # labels are UNSHIFTED; mask prompt/padding with -100 before this call.
    shift_logits = logits[:, :-1, :].contiguous()
    shift_labels = labels[:, 1:].contiguous()
    if not (shift_labels != -100).any():
        raise ValueError('Batch contains no supervised response tokens')
    return F.cross_entropy(shift_logits.reshape(-1, shift_logits.size(-1)),
                           shift_labels.reshape(-1), ignore_index=-100)


def response_logps(logits, labels):
    # Sum conditional token log probabilities over response positions only.
    targets = labels[:, 1:]
    mask = targets != -100
    if not mask.any(dim=-1).all():
        raise ValueError('Each response must contain a supervised token')
    safe_targets = targets.masked_fill(~mask, 0)
    token_logps = logits[:, :-1].log_softmax(-1).gather(
        -1, safe_targets.unsqueeze(-1)
    ).squeeze(-1)
    return (token_logps * mask).sum(-1)


def dpo_loss(policy_chosen_logp, policy_rejected_logp,
             ref_chosen_logp, ref_rejected_logp, beta=0.1):
    if beta <= 0:
        raise ValueError('beta must be positive')
    policy_logratio = policy_chosen_logp - policy_rejected_logp
    ref_logratio = (ref_chosen_logp - ref_rejected_logp).detach()
    return -F.logsigmoid(beta * (policy_logratio - ref_logratio)).mean()


def main():
    torch.manual_seed(42)
    # Tokens 1,2,3 form the prompt; 4,5 are the response, 0 is PAD.
    labels = torch.tensor([[-100, -100, -100, 4, 5, -100]])
    logits = torch.randn(1, 6, 8, requires_grad=True)
    loss = sft_loss(logits, labels)
    loss.backward()
    assert logits.grad[:, :2].abs().sum() == 0
    assert logits.grad[:, 2:4].abs().sum() > 0  # includes prediction of first answer token
    assert logits.grad[:, 4:].abs().sum() == 0
    manual = logits.detach()[0, 2].log_softmax(-1)[4] + logits.detach()[0, 3].log_softmax(-1)[5]
    assert torch.allclose(response_logps(logits.detach(), labels)[0], manual)
    try:
        sft_loss(logits, torch.full_like(labels, -100))
    except ValueError:
        pass
    else:
        raise AssertionError('All-masked targets must be rejected')
    chosen = torch.tensor([-3.0], requires_grad=True)
    rejected = torch.tensor([-4.0], requires_grad=True)
    ref_chosen = torch.tensor([-3.0], requires_grad=True)
    ref_rejected = torch.tensor([-4.0], requires_grad=True)
    dpo = dpo_loss(chosen, rejected, ref_chosen, ref_rejected)
    assert torch.allclose(dpo, torch.log(torch.tensor(2.0)))
    dpo.backward()
    assert chosen.grad.item() < 0 < rejected.grad.item()
    assert ref_chosen.grad is None and ref_rejected.grad is None
    improved = dpo_loss(chosen.detach() + 1, rejected.detach(), ref_chosen, ref_rejected)
    assert improved < dpo
    print('PASS: SFT alignment/masking, response log-probs, DPO reference and gradient direction.')


if __name__ == '__main__':
    main()
